> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `melee-agent debug inspect lifetime-pressure`, a generic, read-only-by-default analyzer for MWCC lifetime pressure and register-allocation mismatches.

**Architecture:** Extend the existing `src.mwcc_debug.pressure_explorer` package instead of creating a parallel workflow. Normalize pcdump/backend-trace facts once, analyze target pressure over those facts, generate source-actionable hypotheses and validation commands, then expose the result through a thin `debug inspect` CLI wrapper.

**Tech Stack:** Python dataclasses, Typer CLI, existing `mwcc_debug` parsers (`colorgraph_parser`, `parser`, `virtual_attribution`, `tiebreak`, `first_divergence`, `role_*`), pytest, existing debug CLI golden-help tests.

def _json_value(value: Any) -> Any:
    if is_dataclass(value):
        return {key: _json_value(val) for key, val in asdict(value).items()}
    if isinstance(value, tuple):
        return [_json_value(item) for item in value]
    if isinstance(value, list):
        return [_json_value(item) for item in value]
    if isinstance(value, dict):
        return {str(key): _json_value(val) for key, val in value.items()}
    if isinstance(value, Path):
        return str(value)
    return value

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from .models import TargetAllocation, TargetSet


def _class_id_from_prefix(prefix: str, default_class_id: int) -> int:
    text = prefix.strip().lower()
    if text in {"0", "gpr", "r"}:
        return 0
    if text in {"1", "fpr", "f"}:
        return 1
    if text == "":
        return default_class_id
    raise ValueError(f"bad register class prefix: {prefix!r}")


def parse_force_phys_spec(raw: str, *, default_class_id: int = 0) -> TargetSet:
    targets: list[TargetAllocation] = []
    for entry in (part.strip() for part in raw.split(",") if part.strip()):
        parts = entry.split(":")
        if len(parts) == 2:
            class_id = default_class_id
            ig_text, phys_text = parts
        elif len(parts) == 3:
            class_id = _class_id_from_prefix(parts[0], default_class_id)
            ig_text, phys_text = parts[1], parts[2]
        else:
            raise ValueError(f"bad force-phys entry: {entry!r}")
        if ig_text[:1].lower() in {"r", "f"}:
            class_id = _class_id_from_prefix(ig_text[0], class_id)
            ig_text = ig_text[1:]
        try:
            targets.append(TargetAllocation(
                class_id=class_id,
                ig_id=int(ig_text),
                expected_phys=int(phys_text),
            ))
        except ValueError as exc:
            raise ValueError(f"bad force-phys entry: {entry!r}") from exc
    if not targets:
        raise ValueError("force-phys spec is empty")
    return TargetSet(function=None, targets=tuple(targets))


def _load_target_mapping(data: dict[str, Any]) -> dict[int, int]:
    mapping = data.get("force_phys") or data.get("virtuals") or {}
    if not isinstance(mapping, dict) or not mapping:
        raise ValueError("target file must contain non-empty force_phys mapping")
    return {int(ig): int(phys) for ig, phys in mapping.items()}


def parse_target_file(path: Path) -> TargetSet:
    data = json.loads(path.read_text())
    function = data.get("function")
    class_id = int(data.get("class_id", data.get("class", 0)))
    targets = tuple(
        TargetAllocation(
            class_id=class_id,
            ig_id=ig,
            expected_phys=phys,
            source=str(path),
        )
        for ig, phys in sorted(_load_target_mapping(data).items())
    )
    return TargetSet(
        function=function,
        targets=targets,
        provenance=dict(data.get("provenance") or {"path": str(path)}),
    )
```

def _source_status(source) -> str:
    if source is None:
